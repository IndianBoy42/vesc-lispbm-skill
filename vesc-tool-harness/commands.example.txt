# Commands for the persistent session. Copy to commands.txt, or append with
#   ./run.sh send '<expression>'      fire and forget
#   ./run.sh ask '<expression>'       send, block for the reply, print it
#   ./run.sh wait '"k":"stats"'       block until a matching LBM| line
# Every line is handled in order; replies appear on the job's stdout.

# A plain expression goes to the REPL.
(set-print-prefix "agent| ")
(get-vin)
(get-rpm)

# Several at once. Split on " ; ".
:multi (get-duty) ; (get-temp-fet 0) ; (get-current)

# Run a whole file, or erase + upload + run a script. Paths relative to this dir.
:run probe.lisp
:upload test.lbm

# One stats sample (also polled automatically at statsIntervalMs).
:stats

# Liveness check: {k:"pong", connected:true}
:ping

# CAN. One attached ESC relays to the nodes behind it; every reply carries
# "can": <id> (-1 = the attached device). Start with a scan.
:can scan
:can 2                      # session target, also sets a print prefix
(get-vin)                   # ...so this runs on node 2
:can 7 (get-temp-fet 0)     # one-shot on node 7, session target unchanged
:can all (get-vin)          # every discovered node, serialised
:can upload 2 probe.lbm     # erase + upload + run on node 2
:can off                    # back to the attached device

# Exit the session. Closes the TCP client cleanly first, so the window's
# single-slot server does not go stale.
# :quit
